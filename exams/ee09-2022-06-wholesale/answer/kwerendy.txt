-- Zapytanie 1: imię, nazwisko i punkty trzech klientów z największą liczbą punktów, malejąco
SELECT imie, nazwisko, punkty FROM klienci ORDER BY punkty DESC LIMIT 3;

-- Zapytanie 2: nazwa typu i liczba klientów przypisanych do każdego typu (relacja)
SELECT typy.nazwa, COUNT(klienci.id) FROM typy JOIN klienci ON klienci.Typy_id = typy.id GROUP BY typy.id, typy.nazwa;

-- Zapytanie 3: zdjecie, imie klienta i jego opinia dla klientów typu 2 i 3 (relacja)
SELECT klienci.zdjecie, klienci.imie, opinie.opinia FROM klienci JOIN opinie ON opinie.Klienci_id = klienci.id WHERE klienci.Typy_id IN (2, 3);

ALTER TABLE klienci ADD adres VARCHAR(50);
