INSERT INTO filmy (gatunki_id, rezyserzy_id, tytul, rok, ocena) VALUES (5, 0, 'Suburbicon', 2017, 5);

-- Zapytanie 2: tytul z tabeli filmy i nazwa gatunku z tabeli gatunki dla filmów z oceną 6, z relacją
SELECT filmy.tytul, gatunki.nazwa FROM filmy JOIN gatunki ON gatunki.id = filmy.gatunki_id WHERE filmy.ocena = 6;

-- Zapytanie 3: tytul i rok dramatów (gatunki_id = 3) z oceną 5
SELECT tytul, rok FROM filmy WHERE gatunki_id = 3 AND ocena = 5;
