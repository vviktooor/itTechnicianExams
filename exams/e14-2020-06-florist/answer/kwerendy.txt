-- Kwerenda 1: nazwa i ulica kwiaciarni z miasta Malbork
SELECT nazwa, ulica FROM kwiaciarnie WHERE miasto = 'Malbork';

-- Kwerenda 2: id zamówienia złożonego 7 stycznia 2017 r.
SELECT id_zam FROM zamowienia WHERE data = '2017-01-07';

-- Kwerenda 3: id_kwiaciarni i liczba zamówień dla poszczególnych kwiaciarni
SELECT id_kwiaciarni, COUNT(*) AS liczba_zamowien FROM zamowienia GROUP BY id_kwiaciarni;
